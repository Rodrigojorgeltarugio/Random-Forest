# python snippet — rode com a venv do projeto from utils import load_file_as_array, extract_features_from_array import numpy as np, os, joblib, pandas as pd path = r"c:\\Users\\rodri\\Downloads\\acelerometro_corrida_100k.csv" arr = load_file_as_array(path) assert arr is not None, "Arquivo não pôde ser lido como dados numéricos (verifique colunas x,y,z ou primeiras 3 colunas numéricas)." feats = extract_features_from_array(arr) X = np.array([list(feats.values())]) # carregar último modelo em ./outputs mods = [f for f in os.listdir('outputs') if f.endswith('.joblib')] if not mods: raise SystemExit('Nenhum modelo .joblib em outputs — treine ou faça upload de um modelo primeiro.') latest = max(mods, key=lambda n: os.path.getmtime(os.path.join('outputs', n))) model_obj = joblib.load(os.path.join('outputs', latest)) clf = model_obj['model'] pred = clf.predict(X)[0] probs = clf.predict_proba(X)[0] if hasattr(clf, 'predict_proba') else None print('prediction (raw):', pred) if probs is not None: # prob da classe prevista idx = list(clf.classes_).index(pred) print('probability:', probs[idx])
